-- ============================================================
-- МЕТОД «НУМЕРАЦИЯ + РАЗНОСТЬ» (Gaps & Islands)
-- ============================================================
-- Суть:     ключ − номер = константа внутри непрерывной группы
-- Разрыв:   пропуск / фильтр / смена категории → разность скачет
-- ------------------------------------------------------------

-- ВИЗУАЛЬНО:
--   id:      5   6   7   8  │ 10  11  12
--   номер:   1   2   3   4  │  5   6   7
--   разн.:   4   4   4   4  │  5   5   5
--            └─ группа A ──┘  └─ группа B ─┘

-- ------------------------------------------------------------
-- СХЕМА:
-- ------------------------------------------------------------
--   последовательность
--         │
--         ▼
--   нумерация (ROW_NUMBER)
--         │
--         ▼
--   разность (ключ − номер)  ← метка группы
--         │
--         ▼
--   одинаковая метка = непрерывность
--   разная метка     = разрыв

-- ------------------------------------------------------------
-- ШАБЛОН (универсальный):
-- ------------------------------------------------------------
WITH numbered AS (
    SELECT
        *,
        key_column - ROW_NUMBER() OVER (ORDER BY key_column) AS grp
    FROM table_name
    WHERE condition = TRUE          -- фильтр ДО нумерации
)
SELECT *
FROM numbered
WHERE grp IN (
    SELECT grp
    FROM numbered
    GROUP BY grp
    HAVING COUNT(*) >= N            -- минимальная длина группы
)
ORDER BY key_column;

-- ------------------------------------------------------------
-- ФОРМУЛА ЗАПОМИНАНИЯ:
-- ------------------------------------------------------------
--   Номер идёт по порядку, ключ идёт с дырками.
--   Их разность — метка непрерывности.
--   Где метка меняется — там разрыв.

-- ------------------------------------------------------------
-- СВОЙСТВА:
-- ------------------------------------------------------------
--   • разность инвариантна внутри группы  → можно GROUP BY
--   • разрыв в ключе или фильтре          → новая группа
--   • не зависит от N                     → любая длина
--   • не требует сортировки результата    → порядок только в OVER

-- ------------------------------------------------------------
-- ПРИМЕНЕНИЕ:
-- ------------------------------------------------------------
--   найти N подряд идущих: даты, id, сессии, транзакции, логи

-- ТЕГИ: #концепция #нумерация-разность #gaps-and-islands
--       #непрерывность #детектор-разрывов
-- ============================================================
