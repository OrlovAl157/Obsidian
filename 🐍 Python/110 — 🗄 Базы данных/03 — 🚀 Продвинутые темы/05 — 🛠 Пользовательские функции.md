# 🛠 Пользовательские функции (UDF) — памятка

> UDF (User Defined Function) — пользовательская функция которая принимает параметры, выполняет логику и возвращает одно значение. Используется как встроенная функция прямо в SELECT, WHERE, SET.

## Справка

| Синтаксис | Что делает |
|---|---|
| `CREATE FUNCTION f(p TYPE) RETURNS TYPE ...` | создать функцию |
| `CREATE OR REPLACE FUNCTION f(...)` | создать или заменить |
| `DROP FUNCTION f` | удалить функцию |
| `DROP FUNCTION IF EXISTS f` | удалить если существует |
| `SHOW FUNCTION STATUS` | список всех функций |
| `SHOW CREATE FUNCTION f` | код функции |
| `DECLARE var TYPE` | объявить локальную переменную |
| `RETURN value` | вернуть значение |
| `SIGNAL SQLSTATE '45000'` | вызвать ошибку |

| Ключевое слово | Значение |
|---|---|
| `DETERMINISTIC` | одинаковые параметры → одинаковый результат |
| `NOT DETERMINISTIC` | может возвращать разные результаты (RAND, NOW) |
| `READS SQL DATA` | читает данные из таблиц |
| `NO SQL` | не обращается к таблицам |
| `MODIFIES SQL DATA` | изменяет данные в таблицах |

---

## 📊 Общая схема

```
DELIMITER //
CREATE FUNCTION function_name(param1 TYPE, param2 TYPE)
RETURNS return_type
DETERMINISTIC       ← или NOT DETERMINISTIC
NO SQL              ← или READS SQL DATA / MODIFIES SQL DATA
BEGIN
    DECLARE var TYPE DEFAULT value;   ← локальные переменные
    -- логика
    RETURN value;                     ← обязательно!
END //
DELIMITER ;

-- Вызов как встроенной функции:
SELECT function_name(arg1, arg2);
SELECT * FROM t WHERE col = function_name(arg);
UPDATE t SET col = function_name(col);
```

## 🟢 Создание функции

**Простейшая функция:**

```sql
DELIMITER //
CREATE FUNCTION IS_EVEN(number INT)
RETURNS INT
DETERMINISTIC
NO SQL
BEGIN
    RETURN IF(number MOD 2 = 0, 1, 0);
END //
DELIMITER ;

SELECT IS_EVEN(10), IS_EVEN(15);
-- Результат: 1 | 0
```

**Функция читающая данные из таблицы:**

```sql
DELIMITER //
CREATE FUNCTION STUDENT_COUNT_BY_GRADE(score INT)
RETURNS INT
NOT DETERMINISTIC
READS SQL DATA
BEGIN
    RETURN (SELECT COUNT(*) FROM Math WHERE grade = score);
END //
DELIMITER ;

SELECT STUDENT_COUNT_BY_GRADE(4), STUDENT_COUNT_BY_GRADE(5);
-- Результат: 2 | 3
```

**Функция с несколькими переменными:**

```sql
DELIMITER //
CREATE FUNCTION MIN_MAX_GRADE()
RETURNS TEXT
NOT DETERMINISTIC
READS SQL DATA
BEGIN
    DECLARE min_grade INT;
    DECLARE max_grade INT;

    SET min_grade := (SELECT MIN(grade) FROM Math);
    SET max_grade := (SELECT MAX(grade) FROM Math);

    RETURN CONCAT('Мин: ', min_grade, ', Макс: ', max_grade);
END //
DELIMITER ;

SELECT MIN_MAX_GRADE();
-- Результат: Мин: 1, Макс: 5
```

## 🔵 Переменные и условия

**DECLARE — локальные переменные:**

```sql
DELIMITER //
CREATE FUNCTION CLASSIFY_GRADE(grade INT)
RETURNS VARCHAR(20)
DETERMINISTIC
NO SQL
BEGIN
    DECLARE result VARCHAR(20);

    IF grade >= 5 THEN
        SET result := 'Отлично';
    ELSEIF grade >= 4 THEN
        SET result := 'Хорошо';
    ELSEIF grade >= 3 THEN
        SET result := 'Удовл.';
    ELSE
        SET result := 'Неудовл.';
    END IF;

    RETURN result;
END //
DELIMITER ;

SELECT CLASSIFY_GRADE(5), CLASSIFY_GRADE(3), CLASSIFY_GRADE(1);
-- Отлично | Удовл. | Неудовл.
```

**CASE внутри функции:**

```sql
DELIMITER //
CREATE FUNCTION DAY_TYPE(weekday_num INT)
RETURNS VARCHAR(10)
DETERMINISTIC
NO SQL
BEGIN
    RETURN CASE
        WHEN weekday_num IN (1, 7) THEN 'Выходной'
        ELSE 'Рабочий'
    END;
END //
DELIMITER ;

SELECT DAY_TYPE(1), DAY_TYPE(3);
-- Выходной | Рабочий
```

**SIGNAL — вызов ошибки:**

```sql
DELIMITER //
CREATE FUNCTION SAFE_DIVIDE(a DECIMAL(10,2), b DECIMAL(10,2))
RETURNS DECIMAL(10,2)
DETERMINISTIC
NO SQL
BEGIN
    IF b = 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Деление на ноль';
    END IF;
    RETURN a / b;
END //
DELIMITER ;

SELECT SAFE_DIVIDE(10, 2);  -- 5.00
SELECT SAFE_DIVIDE(10, 0);  -- Error: Деление на ноль
```

## 🔴 Циклы

**WHILE:**

```sql
DELIMITER //
CREATE FUNCTION SUM_UP_TO_N(n INT)
RETURNS INT
DETERMINISTIC
NO SQL
BEGIN
    DECLARE result INT DEFAULT 0;
    DECLARE i INT DEFAULT 1;

    WHILE i <= n DO
        SET result := result + i;
        SET i := i + 1;
    END WHILE;

    RETURN result;
END //
DELIMITER ;

SELECT SUM_UP_TO_N(5);   -- 1+2+3+4+5 = 15
SELECT SUM_UP_TO_N(10);  -- 55
```

**REPEAT:**

```sql
DELIMITER //
CREATE FUNCTION FACTORIAL(n INT)
RETURNS BIGINT
DETERMINISTIC
NO SQL
BEGIN
    DECLARE result BIGINT DEFAULT 1;
    DECLARE i INT DEFAULT 2;

    REPEAT
        SET result := result * i;
        SET i := i + 1;
    UNTIL i > n END REPEAT;

    RETURN result;
END //
DELIMITER ;

SELECT FACTORIAL(5);   -- 120
SELECT FACTORIAL(10);  -- 3628800
```

**LOOP с LEAVE:**

```sql
DELIMITER //
CREATE FUNCTION POWER_OF_2(n INT)
RETURNS BIGINT
DETERMINISTIC
NO SQL
BEGIN
    DECLARE result BIGINT DEFAULT 1;
    DECLARE i INT DEFAULT 0;

    calc_loop: LOOP
        IF i >= n THEN
            LEAVE calc_loop;
        END IF;
        SET result := result * 2;
        SET i := i + 1;
    END LOOP;

    RETURN result;
END //
DELIMITER ;

SELECT POWER_OF_2(10);  -- 1024
```

## 🟡 Управление функциями

```sql
-- Просмотр всех функций
SHOW FUNCTION STATUS;
SHOW FUNCTION STATUS WHERE Db = 'mydb';

-- Просмотр кода функции
SHOW CREATE FUNCTION IS_EVEN;

-- Удалить функцию
DROP FUNCTION IS_EVEN;

-- Удалить если существует
DROP FUNCTION IF EXISTS IS_EVEN;

-- Заменить существующую (MySQL 8.0+)
CREATE OR REPLACE FUNCTION IS_EVEN(n INT)
RETURNS INT DETERMINISTIC NO SQL
RETURN IF(n MOD 2 = 0, 1, 0);
```

## 🟣 Функция vs Процедура

| Критерий | Функция | Процедура |
|---|---|---|
| Возвращает | одно значение (`RETURN`) | нет или через `OUT` параметры |
| Вызов | в `SELECT`, `WHERE`, `SET` | только `CALL` |
| Использование в запросе | ✅ `SELECT func()` | ❌ нельзя |
| Несколько результатов | ❌ | ✅ через `SELECT` внутри |
| Транзакции | ❌ нет | ✅ можно |
| Когда использовать | вычисление значения | сложная логика, изменение данных |

## ⚡ Быстрые примеры

```sql
-- Проверка чётности
CREATE FUNCTION IS_EVEN(n INT) RETURNS INT DETERMINISTIC NO SQL
RETURN IF(n MOD 2 = 0, 1, 0);

-- Полное имя
CREATE FUNCTION FULL_NAME(first VARCHAR(50), last VARCHAR(50))
RETURNS VARCHAR(100) DETERMINISTIC NO SQL
RETURN CONCAT(first, ' ', last);

-- Возраст из даты рождения
CREATE FUNCTION AGE(birth_date DATE)
RETURNS INT NOT DETERMINISTIC NO SQL
RETURN TIMESTAMPDIFF(YEAR, birth_date, CURDATE());

-- Скидка
CREATE FUNCTION APPLY_DISCOUNT(price DECIMAL(10,2), pct INT)
RETURNS DECIMAL(10,2) DETERMINISTIC NO SQL
RETURN ROUND(price * (1 - pct / 100.0), 2);

-- Использование в запросах
SELECT name, FULL_NAME(first_name, last_name), AGE(birth_date)
FROM Users;

SELECT title, price, APPLY_DISCOUNT(price, 20) AS discounted
FROM Books;

SELECT * FROM Books WHERE IS_EVEN(id) = 1;
```

## 💡 Практические замечания

- Функция всегда должна содержать `RETURN` — без него ошибка
- `DETERMINISTIC` — подсказка оптимизатору для кэширования результата
- `NOT DETERMINISTIC` — если функция зависит от текущего времени, случайности или данных в таблице
- `READS SQL DATA` обязательно если функция делает `SELECT` из таблиц
- Функция не может делать `INSERT/UPDATE/DELETE` — для этого процедура
- `DELIMITER //` обязателен если тело функции многострочное

## ⚠️ Частые ошибки

**❌ Забыл RETURN:**
```sql
CREATE FUNCTION bad(n INT) RETURNS INT DETERMINISTIC NO SQL
BEGIN
    DECLARE x INT DEFAULT n * 2;
    -- нет RETURN! → ошибка
END;

-- ✅
BEGIN
    RETURN n * 2;
END;
```

**❌ Неверный атрибут для функции с SELECT:**
```sql
CREATE FUNCTION get_count() RETURNS INT
DETERMINISTIC NO SQL   -- ❌ NO SQL, но есть SELECT из таблицы!
BEGIN
    RETURN (SELECT COUNT(*) FROM Books);
END;

-- ✅
NOT DETERMINISTIC READS SQL DATA
```

**❌ Вызов через CALL:**
```sql
CALL IS_EVEN(10);       -- ❌ CALL только для процедур!
SELECT IS_EVEN(10);     -- ✅
```

---

## ✅ Главные правила

✅ Функция всегда возвращает одно значение через `RETURN`  
✅ Можно использовать в `SELECT`, `WHERE`, `SET` как встроенную функцию  
✅ `DETERMINISTIC` — если результат зависит только от аргументов  
✅ `READS SQL DATA` — если есть `SELECT` из таблиц  
✅ `DECLARE` — объявить локальные переменные в начале `BEGIN`  
✅ `SIGNAL SQLSTATE '45000'` — вызвать пользовательскую ошибку  
✅ `DROP FUNCTION IF EXISTS` — безопасное удаление  
✅ `DELIMITER //` — обязателен для многострочного тела
