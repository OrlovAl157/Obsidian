-- ============================================================
-- МЕТОД «УСЛОВНАЯ АГРЕГАЦИЯ» (Pivot через CASE/IF)
-- ============================================================
-- Суть:     внутри агрегата — условие, снаружи — GROUP BY
-- Эффект:   длинная таблица → широкая, без JOIN и без подзапросов
-- ------------------------------------------------------------

-- ВИЗУАЛЬНО:
--   Было (длинная):              Стало (широкая):
--   id | revenue | month         id | Jan  | Feb  | Mar
--   1  | 8000    | Jan           1  | 8000 | 7000 | 6000
--   1  | 7000    | Feb
--   1  | 6000    | Mar

-- ------------------------------------------------------------
-- СХЕМА:
-- ------------------------------------------------------------
--   длинная таблица
--         │
--         ▼
--   IF(условие, значение, NULL)   ← для КАЖДОЙ строки отдельно
--         │
--         ▼
--   SUM / MAX / MIN (...)          ← агрегат игнорирует NULL
--         │
--         ▼
--   GROUP BY ключ                  ← одна строка на ключ
--         │
--         ▼
--   значение из нужной строки «всплыло» в свою колонку

-- ------------------------------------------------------------
-- ШАБЛОН (универсальный):
-- ------------------------------------------------------------
SELECT
    key_column,
    SUM(IF(category_column = 'A', value_column, NULL)) AS A_value,
    SUM(IF(category_column = 'B', value_column, NULL)) AS B_value,
    SUM(IF(category_column = 'C', value_column, NULL)) AS C_value
FROM table_name
GROUP BY key_column;

-- ------------------------------------------------------------
-- ФОРМУЛА ЗАПОМИНАНИЯ:
-- ------------------------------------------------------------
--   Внутри — IF/CASE отбирает «своё» значение, остальным даёт NULL.
--   Снаружи — SUM/MAX/MIN съедает все NULL и оставляет единственное число.
--   GROUP BY сводит построчный результат в одну строку на ключ.

-- ------------------------------------------------------------
-- СВОЙСТВА:
-- ------------------------------------------------------------
--   • IF/CASE работает ПОСТРОЧНО, до агрегации и до GROUP BY
--   • SUM/MAX/MIN игнорируют NULL — поэтому «выживает» ровно 1 значение
--   • работает только если на пару (ключ, категория) ≤ 1 строка
--     (иначе агрегат схлопнет несколько значений в одно, см. ONLY_FULL_GROUP_BY)
--   • вместо SUM можно MAX/MIN — если в паре гарантированно 1 значение,
--     результат совпадёт; SUM удобнее, если нужно именно сложение

-- ------------------------------------------------------------
-- ПРИМЕНЕНИЕ:
-- ------------------------------------------------------------
--   длинная таблица → широкая без JOIN: выручка по месяцам,
--   оценки по предметам, статусы по этапам — pivot одним запросом

-- ============================================================
-- ТЕГИ: #концепция #условная-агрегация #pivot
--       #sum-if #group-by #длинная-в-широкую
-- ============================================================
